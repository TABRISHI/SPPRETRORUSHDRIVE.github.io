// DEMOLITION DERBY - Standalone Three.js Game // Controls: WASD / Arrow Keys // Last car alive wins! const scene = new THREE.Scene(); scene.background = new THREE.Color(0x151515); const camera = new THREE.PerspectiveCamera( 65, innerWidth / innerHeight, 0.1, 500 ); const renderer = new THREE.WebGLRenderer({ antialias: true }); renderer.setSize(innerWidth, innerHeight); document.body.appendChild(renderer.domElement); // ---------------- LIGHTING ---------------- scene.add(new THREE.HemisphereLight(0xffffff, 0x333333, 2)); const light = new THREE.DirectionalLight(0xffffff, 2); light.position.set(50, 100, 50); scene.add(light); // ---------------- ARENA ---------------- const arenaSize = 100; const floor = new THREE.Mesh( new THREE.CylinderGeometry(arenaSize, arenaSize, 1, 64), new THREE.MeshStandardMaterial({ color: 0x444444 }) ); floor.position.y = -0.5; scene.add(floor); // Arena wall const wall = new THREE.Mesh( new THREE.TorusGeometry(arenaSize, 2, 8, 64), new THREE.MeshStandardMaterial({ color: 0xff8800 }) ); wall.rotation.x = Math.PI / 2; scene.add(wall); // ---------------- CAR CREATION ---------------- const cars = []; function createCar(color, name, isPlayer = false) { const group = new THREE.Group(); // Body const body = new THREE.Mesh( new THREE.BoxGeometry(4, 1.2, 7), new THREE.MeshStandardMaterial({ color: color, metalness: 0.4 }) ); body.position.y = 1; group.add(body); // Cabin const cabin = new THREE.Mesh( new THREE.BoxGeometry(3, 1, 3), new THREE.MeshStandardMaterial({ color: 0x111111 }) ); cabin.position.y = 1.9; group.add(cabin); // Wheels for (const x of [-2.1, 2.1]) { for (const z of [-2.2, 2.2]) { const wheel = new THREE.Mesh( new THREE.CylinderGeometry(0.7, 0.7, 0.5, 12), new THREE.MeshStandardMaterial({ color: 0x111111 }) ); wheel.rotation.z = Math.PI / 2; wheel.position.set(x, 0.6, z); group.add(wheel); } } scene.add(group); const car = { mesh: group, name: name, health: 100, speed: 0, velocity: new THREE.Vector3(), angle: Math.random() * Math.PI * 2, player: isPlayer, alive: true, cooldown: 0 }; cars.push(car); return car; } // ---------------- CREATE CARS ---------------- const player = createCar(0x2196ff, "YOU", true); player.mesh.position.set(0, 0, -60); for (let i = 0; i < 7; i++) { const angle = (i / 7) * Math.PI * 2; const car = createCar( [ 0xff3333, 0x33ff66, 0xffff33, 0xff33ff, 0xff8833, 0x33ffff, 0xffffff ][i], "BOT " + (i + 1) ); car.mesh.position.set( Math.cos(angle) * 55, 0, Math.sin(angle) * 55 ); } // ---------------- CONTROLS ---------------- const keys = {}; addEventListener("keydown", e => { keys[e.key.toLowerCase()] = true; }); addEventListener("keyup", e => { keys[e.key.toLowerCase()] = false; }); // ---------------- PHYSICS ---------------- function drivePlayer(car) { if (!car.alive) return; const forward = new THREE.Vector3( Math.sin(car.angle), 0, Math.cos(car.angle) ); if (keys.w || keys.arrowup) car.velocity.addScaledVector(forward, 0.5); if (keys.s || keys.arrowdown) car.velocity.addScaledVector(forward, -0.3); if (keys.a || keys.arrowleft) car.angle += 0.055; if (keys.d || keys.arrowright) car.angle -= 0.055; car.velocity.multiplyScalar(0.96); if (car.velocity.length() > 1.2) car.velocity.setLength(1.2); car.mesh.position.add(car.velocity); car.mesh.rotation.y = car.angle; keepInsideArena(car); } // ---------------- AI ---------------- function driveBot(car) { if (!car.alive) return; // Find closest living car let target = player; let best = Infinity; for (const other of cars) { if (other === car || !other.alive) continue; const distance = car.mesh.position.distanceTo(other.mesh.position); if (distance < best) { best = distance; target = other; } } const direction = target.mesh.position .clone() .sub(car.mesh.position); const wantedAngle = Math.atan2(direction.x, direction.z); let difference = wantedAngle - car.angle; while (difference > Math.PI) difference -= Math.PI * 2; while (difference < -Math.PI) difference += Math.PI * 2; car.angle += Math.sign(difference) * 0.025; const forward = new THREE.Vector3( Math.sin(car.angle), 0, Math.cos(car.angle) ); car.velocity.addScaledVector(forward, 0.28); car.velocity.multiplyScalar(0.97); if (car.velocity.length() > 1) car.velocity.setLength(1); car.mesh.position.add(car.velocity); car.mesh.rotation.y = car.angle; keepInsideArena(car); } // ---------------- ARENA COLLISION ---------------- function keepInsideArena(car) { const distance = Math.sqrt( car.mesh.position.x ** 2 + car.mesh.position.z ** 2 ); if (distance > arenaSize - 5) { const angle = Math.atan2( car.mesh.position.z, car.mesh.position.x ); car.mesh.position.x = Math.cos(angle) * (arenaSize - 5); car.mesh.position.z = Math.sin(angle) * (arenaSize - 5); // Bounce away from wall car.velocity.multiplyScalar(-0.7); damage(car, 3); } } // ---------------- CAR COLLISIONS ---------------- function collisions() { for (let i = 0; i < cars.length; i++) { const a = cars[i]; if (!a.alive) continue; for (let j = i + 1; j < cars.length; j++) { const b = cars[j]; if (!b.alive) continue; const distance = a.mesh.position.distanceTo(b.mesh.position); if (distance < 5) { const impact = a.velocity.length() + b.velocity.length(); // Push cars apart const direction = a.mesh.position .clone() .sub(b.mesh.position) .normalize(); a.velocity.addScaledVector(direction, 0.5); b.velocity.addScaledVector(direction, -0.5); // Damage if (impact > 0.4) { damage(a, impact * 4); damage(b, impact * 4); // Small collision spin a.angle += (Math.random() - 0.5) * 0.3; b.angle += (Math.random() - 0.5) * 0.3; } } } } } // ---------------- DAMAGE ---------------- function damage(car, amount) { if (!car.alive || car.cooldown > 0) return; car.health -= amount; car.cooldown = 0.1; if (car.health <= 0) { car.health = 0; car.alive = false; // Destroy/disable car car.mesh.rotation.z = (Math.random() - 0.5) * 1.5; car.mesh.position.y = 0.5; console.log(car.name + " ELIMINATED!"); } } // ---------------- HUD ---------------- const hud = document.createElement("div"); hud.style.cssText = ` position:fixed; top:20px; left:20px; color:white; font-family:monospace; font-size:18px; background:rgba(0,0,0,.7); padding:15px; z-index:10; `; document.body.appendChild(hud); function updateHUD() { const alive = cars.filter(c => c.alive); hud.innerHTML = ` DEMOLITION DERBY

YOUR HP: ${Math.ceil(player.health)}
CARS LEFT: ${alive.length}

WASD / ARROWS = DRIVE `; if (!player.alive) { hud.innerHTML += `
YOU WERE ELIMINATED! `; } else if (alive.length === 1) { hud.innerHTML += `
🏆 YOU WIN! `; } } // ---------------- CAMERA ---------------- function updateCamera() { const target = player.mesh.position.clone(); const behind = new THREE.Vector3( -Math.sin(player.angle) * 18, 12, -Math.cos(player.angle) * 18 ); camera.position.lerp( target.clone().add(behind), 0.08 ); camera.lookAt( target.clone().add(new THREE.Vector3(0, 1, 0)) ); } // ---------------- GAME LOOP ---------------- function gameLoop() { requestAnimationFrame(gameLoop); for (const car of cars) { car.cooldown -= 0.016; if (car.player) drivePlayer(car); else driveBot(car); } collisions(); updateCamera(); updateHUD(); renderer.render(scene, camera); } gameLoop(); // ---------------- RESIZE ---------------- addEventListener("resize", () => { camera.aspect = innerWidth / innerHeight; camera.updateProjectionMatrix(); renderer.setSize( innerWidth, innerHeight ); });
