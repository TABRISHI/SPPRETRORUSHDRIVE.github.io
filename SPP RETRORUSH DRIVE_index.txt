  function trackNormal(i){
    const t = trackTangents[i];
    return new THREE.Vector3(-t.z, 0, t.x).normalize();
  }


  // road surface texture: asphalt + dashed centerline + white edge lines
  function makeRoadTexture(){
    const c = document.createElement('canvas');
    c.width = 128; c.height = 512;
    const ctx = c.getContext('2d');
    ctx.fillStyle = '#3a3b3f';
    ctx.fillRect(0,0,128,512);
    ctx.fillStyle = 'rgba(255,255,255,0.85)';
    ctx.fillRect(8,0,5,512);
    ctx.fillRect(115,0,5,512);
    ctx.fillStyle = 'rgba(255,220,120,0.9)';
    for(let y=0;y<512;y+=48){ ctx.fillRect(60,y,8,26); }
    const tex = new THREE.CanvasTexture(c);
    tex.wrapS = THREE.ClampToEdgeWrapping;
    tex.wrapT = THREE.RepeatWrapping;
    tex.repeat.set(1, TOTAL_LEN/10);
    return tex;
  }


  function makeTree(x,z){
    const g = new THREE.Group();
    const trunk = new THREE.Mesh(new THREE.CylinderGeometry(0.25,0.3,1.6,8), new THREE.MeshStandardMaterial({color:0x6b4226}));
    trunk.position.y = 0.8; trunk.castShadow = true;

  function makePalm(x,z){
    const g = new THREE.Group();
    const trunk = new THREE.Mesh(new THREE.CylinderGeometry(0.18,0.28,3.4,8), new THREE.MeshStandardMaterial({color:0x8a6b45}));
    trunk.position.y = 1.7; trunk.rotation.z = 0.08; trunk.castShadow = true;
    g.add(trunk);
    const frondMat = new THREE.MeshStandardMaterial({color:0x3f9a4f});
    for(let i=0;i<6;i++){
      const pivot = new THREE.Group();

      pivot.rotation.y = (i/6)*Math.PI*2;

      frond.position.set(0.85, -0.15, 0);
      frond.rotation.z = -0.35;

      pivot.add(frond);
      g.add(pivot);

    trunk.position.y = 1.15; trunk.castShadow = true;
    g.add(trunk);

  function makeDecoration(x,z,type){
    if(type === 'palm') return makePalm(x,z);
    if(type === 'cactus') return makeCactus(x,z);
    return makeTree(x,z);
  }
  function nearestTrackDist(x,z){
    let best = Infinity;
    for(let i=0;i<SEGMENTS;i+=4){
      const dx = trackPts[i].x-x, dz = trackPts[i].z-z;
      const d = dx*dx+dz*dz;
      if(d<best) best = d;
    }
    return Math.sqrt(best);
  }


  // Low-poly spectators — grouped into a handful of InstancedMesh batches by shirt
  // color, so a crowd of dozens still only costs a few draw calls.
  const CROWD_SHIRT_COLORS = [0xd6432f, 0x2f6fd6, 0xe0c020, 0x2f9e52, 0x8a3fbf, 0xdddddd, 0x333333];
  const crowdHeadGeo = new THREE.SphereGeometry(0.2, 7, 7);
  const crowdBodyGeo = new THREE.CylinderGeometry(0.21, 0.26, 0.85, 7);
  const crowdHeadMat = new THREE.MeshStandardMaterial({ color:0xd9a679 });
  function addCrowd(parentGroup, rows, cols, spacingX, spacingZ, baseX, baseY, baseZ){
    const batches = {};
    const dummy = new THREE.Object3D();
    for(let r=0;r<rows;r++){
      for(let c=0;c<cols;c++){
        const color = CROWD_SHIRT_COLORS[Math.floor(Math.random()*CROWD_SHIRT_COLORS.length)];
        if(!batches[color]) batches[color] = [];
        const x = baseX + (c-(cols-1)/2)*spacingX + (Math.random()-0.5)*0.15;
        const y = baseY; // flat ground-level crowd, rows offset by depth (z) only
        const z = baseZ + r*spacingZ;
        batches[color].push({x,y,z});
      }
    }
    Object.keys(batches).forEach(colorKey=>{
      const list = batches[colorKey];
      const bodyMesh = new THREE.InstancedMesh(crowdBodyGeo, new THREE.MeshStandardMaterial({color:Number(colorKey)}), list.length);
      const headMesh = new THREE.InstancedMesh(crowdHeadGeo, crowdHeadMat, list.length);
      list.forEach((p,i)=>{
        dummy.position.set(p.x, p.y+0.43, p.z);
        dummy.updateMatrix();
        bodyMesh.setMatrixAt(i, dummy.matrix);
        dummy.position.set(p.x, p.y+1.05, p.z);
        dummy.updateMatrix();
        headMesh.setMatrixAt(i, dummy.matrix);
      });
      bodyMesh.instanceMatrix.needsUpdate = true;
      headMesh.instanceMatrix.needsUpdate = true;
      parentGroup.add(bodyMesh, headMesh);
    });
  }


  // Vertical gradient sky per map, rendered as a screen-space background texture
  // instead of a flat color — gives each circuit a proper horizon instead of a
  // uniform wash of color.
  function mixHex(a,b,t){
    const ar=(a>>16)&255, ag=(a>>8)&255, ab=a&255;
    const br=(b>>16)&255, bg=(b>>8)&255, bb=b&255;
    const r=Math.round(ar+(br-ar)*t), g=Math.round(ag+(bg-ag)*t), bl=Math.round(ab+(bb-ab)*t);
    return (r<<16)|(g<<8)|bl;
  }
  function makeSkyTexture(zenithHex, horizonHex){
    const c = document.createElement('canvas');
    c.width = 2; c.height = 256;
    const ctx = c.getContext('2d');
    const grad = ctx.createLinearGradient(0,0,0,256);
    grad.addColorStop(0, '#'+zenithHex.toString(16).padStart(6,'0'));
    grad.addColorStop(0.7, '#'+horizonHex.toString(16).padStart(6,'0'));
    grad.addColorStop(1, '#'+mixHex(horizonHex,0xffffff,0.3).toString(16).padStart(6,'0'));
    ctx.fillStyle = grad;
    ctx.fillRect(0,0,2,256);
    return new THREE.CanvasTexture(c);
  }


  const barrierGeo = new THREE.BoxGeometry(0.3, 0.7, 3.2);
  const barrierRed = new THREE.MeshStandardMaterial({ color:0xcc2b2b });
  const barrierWhite = new THREE.MeshStandardMaterial({ color:0xe8e8e8 });
  let oceanMesh = null;


  function buildWorld(mapIndex){
    const preset = MAP_PRESETS[mapIndex];
    currentMapIndex = mapIndex;
    TRACK_WIDTH = preset.width;
    setMusicTheme(mapIndex); // keep the soundtrack in sync with whichever circuit is actually loaded


    // tear down the previous map's meshes, if any
    if(roadMesh){ scene.remove(roadMesh); roadMesh.geometry.dispose(); roadMesh.material.dispose(); }
    if(startStripeMesh){ scene.remove(startStripeMesh); startStripeMesh.geometry.dispose(); startStripeMesh.material.dispose(); }
    barrierMeshes.forEach(b=> scene.remove(b));
    barrierMeshes = [];
    if(sceneryGroup) scene.remove(sceneryGroup);
    if(grandstandGroup) scene.remove(grandstandGroup);
    if(crowdGroup) scene.remove(crowdGroup);
    if(oceanMesh){ scene.remove(oceanMesh); oceanMesh.geometry.dispose(); oceanMesh.material.dispose(); oceanMesh = null; }


    // theme this map: ground color, gradient sky, fog, and mountain tint all change per circuit
    ground.material.color.setHex(preset.groundColor);
    if(scene.background && scene.background.isTexture) scene.background.dispose();
    scene.background = makeSkyTexture(preset.skyColor, mixHex(preset.skyColor, preset.fogColor, 0.5));
    scene.fog.color.setHex(preset.fogColor);
    mtnMat.color.setHex(preset.mountainColor);


    const rawPts = preset.points.map(p => new THREE.Vector3(p[0]*preset.scale, 0, p[1]*preset.scale));
    curve = new THREE.CatmullRomCurve3(rawPts, true, 'catmullrom', 0.5);


    trackPts = []; trackTangents = [];
    for(let i=0;i<SEGMENTS;i++){
      const t = i/SEGMENTS;
      trackPts.push(curve.getPointAt(t));
      trackTangents.push(curve.getTangentAt(t).normalize());
    }
    cumLen = [0];
    for(let i=1;i<SEGMENTS;i++) cumLen.push(cumLen[i-1] + trackPts[i].distanceTo(trackPts[i-1]));
    TOTAL_LEN = cumLen[SEGMENTS-1] + trackPts[0].distanceTo(trackPts[SEGMENTS-1]);


    const roadTex = makeRoadTexture();
    const roadPositions = [], roadUVs = [], roadIndices = [];
    for(let i=0;i<SEGMENTS;i++){
      const n = trackNormal(i);
      const p = trackPts[i];
      const left = p.clone().addScaledVector(n, TRACK_WIDTH/2);
      const right = p.clone().addScaledVector(n, -TRACK_WIDTH/2);
      roadPositions.push(left.x, 0.02, left.z, right.x, 0.02, right.z);
      const v = cumLen[i]/10;
      roadUVs.push(0, v, 1, v);
    }
    for(let i=0;i<SEGMENTS;i++){
      const a = i*2, b = i*2+1, c2 = ((i+1)%SEGMENTS)*2, d = ((i+1)%SEGMENTS)*2+1;
      roadIndices.push(a,c2,b, b,c2,d);
    }
    const roadGeo = new THREE.BufferGeometry();
    roadGeo.setAttribute('position', new THREE.Float32BufferAttribute(roadPositions,3));
    roadGeo.setAttribute('uv', new THREE.Float32BufferAttribute(roadUVs,2));
    roadGeo.setIndex(roadIndices);
    roadGeo.computeVertexNormals();
    roadMesh = new THREE.Mesh(roadGeo, new THREE.MeshStandardMaterial({ map: roadTex, roughness:0.95 }));
    roadMesh.receiveShadow = true;
    scene.add(roadMesh);


    // start/finish checker strip at index 0
    const c = document.createElement('canvas');
    c.width = c.height = 64;
    const ctx = c.getContext('2d');
    for(let y=0;y<8;y++) for(let x=0;x<8;x++){
      ctx.fillStyle = (x+y)%2===0 ? '#fff' : '#111';
      ctx.fillRect(x*8,y*8,8,8);
    }
    const stripeTex = new THREE.CanvasTexture(c);
    const n0 = trackNormal(0);
    const p0 = trackPts[0];
    startStripeMesh = new THREE.Mesh(new THREE.PlaneGeometry(TRACK_WIDTH, 4), new THREE.MeshStandardMaterial({ map: stripeTex }));
    startStripeMesh.rotation.x = -Math.PI/2;
    startStripeMesh.rotation.z = -Math.atan2(trackTangents[0].x, trackTangents[0].z);
    startStripeMesh.position.set(p0.x, 0.03, p0.z);
    scene.add(startStripeMesh);


    // guardrail barriers along both edges
    for(let i=0;i<SEGMENTS;i+=5){
      const n = trackNormal(i);
      const p = trackPts[i];
      const ang = Math.atan2(trackTangents[i].x, trackTangents[i].z);
      const mat = (Math.floor(i/5)%2===0) ? barrierRed : barrierWhite;
      [1,-1].forEach(side=>{
        const b = new THREE.Mesh(barrierGeo, mat);
        b.position.copy(p).addScaledVector(n, side*(TRACK_WIDTH/2+0.6));
        b.position.y = 0.35;
        b.rotation.y = ang;
        b.castShadow = true;
        scene.add(b);
        barrierMeshes.push(b);
      });
    }


    // compute where the ocean will start (if this map has one) so scenery
    // placement below can avoid scattering trees out on top of the water
    let oceanStart = Infinity;
    if(preset.ocean){
      let maxX = -Infinity;
      for(let i=0;i<SEGMENTS;i++) if(trackPts[i].x > maxX) maxX = trackPts[i].x;
      oceanStart = maxX + 45;
    }


    // scenery: trees/palms/cacti scattered outside the track corridor, themed per map
    sceneryGroup = new THREE.Group();
    let placed = 0, attempts = 0;
    while(placed < 100 && attempts < 1400){
      attempts++;
      const x = (Math.random()-0.5)*760;
      const z = (Math.random()-0.5)*620;
      if(nearestTrackDist(x,z) > TRACK_WIDTH/2 + 9 && x < oceanStart - 15){
        sceneryGroup.add(makeDecoration(x,z,preset.decoration));
        placed++;
      }
    }
    scene.add(sceneryGroup);


    // coastal maps get an ocean along one side of the track — start it well past
    // the track's own footprint so it never overlaps the road
    if(preset.ocean){
      const oceanWidth = 380;
      oceanMesh = new THREE.Mesh(
        new THREE.PlaneGeometry(oceanWidth, 1400),
        new THREE.MeshStandardMaterial({ color:0x2b7fae, roughness:0.25, metalness:0.15 })
      );
      oceanMesh.rotation.x = -Math.PI/2;
      oceanMesh.position.set(oceanStart + oceanWidth/2, 0.015, 0);
      scene.add(oceanMesh);
    }


    // grandstands near start/finish — one on each side of the track, packed with fans
    grandstandGroup = new THREE.Group();
    const nG = trackNormal(0), pG = trackPts[0];
    [1, -1].forEach(side=>{
      const stand = new THREE.Group();
      const base = new THREE.Mesh(new THREE.BoxGeometry(30,5,6), new THREE.MeshStandardMaterial({color:0x555c66}));
      base.position.y = 2.5; base.castShadow = true;
      const roof = new THREE.Mesh(new THREE.BoxGeometry(31,0.6,7), new THREE.MeshStandardMaterial({color:0x2b2f36}));
      roof.position.y = 5.3;
      const stripe = new THREE.Mesh(new THREE.BoxGeometry(30,0.8,6.2), new THREE.MeshStandardMaterial({color:0xffb020}));
      stripe.position.y = 4.6;
      stand.add(base, roof, stripe);
      stand.position.copy(pG).addScaledVector(nG, side*(TRACK_WIDTH/2+22));
      stand.rotation.y = Math.atan2(trackTangents[0].x, trackTangents[0].z);
      grandstandGroup.add(stand);
      addCrowd(stand, 3, 18, 1.6, 0.65, 0, 0, -3.6);
    });
    scene.add(grandstandGroup);


    // crowd: dense trackside clusters scattered all the way around the loop, so
    // there are always people visible while you're actually driving, not just at
    // the start/finish grandstands
    crowdGroup = new THREE.Group();
    const CLUSTER_COUNT = 10;
    for(let k=0;k<CLUSTER_COUNT;k++){
      const idx = Math.floor(((k+0.5)/CLUSTER_COUNT) * SEGMENTS);
      const nc = trackNormal(idx), pc = trackPts[idx];
      const cluster = new THREE.Group();
      addCrowd(cluster, 2, 6, 0.75, 0.6, 0, 0, 0);
      cluster.position.copy(pc).addScaledVector(nc, (Math.random()<0.5?1:-1) * (TRACK_WIDTH/2 + 2.2));
      cluster.rotation.y = Math.atan2(trackTangents[idx].x, trackTangents[idx].z);
      crowdGroup.add(cluster);
    }
    scene.add(crowdGroup);
  }


  buildWorld(0); // default map on first load

  function buildCar(bodyColor){
    // Low-poly Aventador-style silhouette: wedge nose, cab-backward greenhouse, wide
    // rear haunches, big fixed wing, Y-shaped taillights, angular diamond headlights.
    const car = new THREE.Group();
    const bodyMat = new THREE.MeshStandardMaterial({ color:bodyColor, metalness:0.55, roughness:0.25 });
    const glassMat = new THREE.MeshPhysicalMaterial({ color:0x141b22, metalness:0.1, roughness:0.05, transparent:true, opacity:0.6 });
    const trimMat = new THREE.MeshStandardMaterial({ color:0x0c0d0f, metalness:0.5, roughness:0.35 });
    const wheelMat = new THREE.MeshStandardMaterial({ color:0x0d0d0d, roughness:0.85 });
    const hubMat = new THREE.MeshStandardMaterial({ color:0xd8d8d8, metalness:0.8, roughness:0.25 });
    const lightMat = new THREE.MeshStandardMaterial({ color:0xfff5c0, emissive:0xfff5c0, emissiveIntensity:0.9 });
    const tailMat = new THREE.MeshStandardMaterial({ color:0xff2b2b, emissive:0xff2b2b, emissiveIntensity:0.6 });


    // low, wide monocoque tub
    const tub = new THREE.Mesh(new THREE.BoxGeometry(2.05, 0.34, 4.5), bodyMat);
    tub.position.y = 0.3; tub.castShadow = true; car.add(tub);


    // wedge nose — tapered and sloped downward like the Aventador's low front
    const nose = new THREE.Mesh(new THREE.BoxGeometry(1.75, 0.24, 1.1), bodyMat);
    nose.position.set(0, 0.34, 2.35);
    nose.rotation.x = -0.16;
    nose.castShadow = true; car.add(nose);


    // cab-backward greenhouse, set low and toward the rear
    const cabin = new THREE.Mesh(new THREE.BoxGeometry(1.5, 0.34, 1.5), bodyMat);
    cabin.position.set(0, 0.66, -0.55); cabin.castShadow = true; car.add(cabin);


    const windshield = new THREE.Mesh(new THREE.BoxGeometry(1.4, 0.4, 1.0), glassMat);
    windshield.position.set(0, 0.86, 0.35);
    windshield.rotation.x = 0.32;
    windshield.castShadow = true; car.add(windshield);


    const roofTrim = new THREE.Mesh(new THREE.BoxGeometry(1.42, 0.06, 1.5), trimMat);
    roofTrim.position.set(0, 0.86, -0.55); car.add(roofTrim);


    // flared rear haunches / engine deck
    const rearDeck = new THREE.Mesh(new THREE.BoxGeometry(2.1, 0.3, 1.5), bodyMat);
    rearDeck.position.set(0, 0.5, -1.7); rearDeck.castShadow = true; car.add(rearDeck);
    const engineLouvers = new THREE.Mesh(new THREE.BoxGeometry(1.2, 0.05, 1.1), trimMat);
    engineLouvers.position.set(0, 0.66, -1.7); car.add(engineLouvers);


    // fixed rear wing, mounted high on twin struts
    const wing = new THREE.Mesh(new THREE.BoxGeometry(1.9, 0.06, 0.55), trimMat);
    wing.position.set(0, 1.05, -2.55);
    const strutGeo = new THREE.BoxGeometry(0.08, 0.42, 0.1);
    const strutL = new THREE.Mesh(strutGeo, trimMat); strutL.position.set(-0.75, 0.82, -2.5);
    const strutR = new THREE.Mesh(strutGeo, trimMat); strutR.position.set(0.75, 0.82, -2.5);
    car.add(wing, strutL, strutR);


    // rear diffuser
    const diffuser = new THREE.Mesh(new THREE.BoxGeometry(1.8, 0.12, 0.4), trimMat);
    diffuser.position.set(0, 0.16, -2.35);
    diffuser.rotation.x = 0.1;
    car.add(diffuser);


    // side skirts / intake scoops ahead of the rear wheels
    [-1.03, 1.03].forEach(x=>{
      const skirt = new THREE.Mesh(new THREE.BoxGeometry(0.14, 0.16, 1.2), trimMat);
      skirt.position.set(x, 0.28, -0.3);
      car.add(skirt);
    });


    // mirrors
    [[-1.05,0.78,0.75],[1.05,0.78,0.75]].forEach(p=>{
      const mirror = new THREE.Mesh(new THREE.BoxGeometry(0.12,0.12,0.26), trimMat);
      mirror.position.set(p[0],p[1],p[2]);
      car.add(mirror);
    });


    // exhaust pipes, centered like the Aventador's hexagonal outlets
    [[-0.22,0.24,2.55],[0.22,0.24,2.55]].forEach(p=>{
      const pipe = new THREE.Mesh(new THREE.CylinderGeometry(0.09,0.09,0.22,6), hubMat);
      pipe.rotation.x = Math.PI/2;
      pipe.position.set(p[0],p[1],p[2]);
      car.add(pipe);
    });


    // wheels: wide stance, wheels pushed toward the corners
    const wheelGeo = new THREE.CylinderGeometry(0.46, 0.46, 0.4, 16);
    const hubGeo = new THREE.CylinderGeometry(0.18, 0.18, 0.42, 5);
    function makeWheel(){
      const g = new THREE.Group();
      const w = new THREE.Mesh(wheelGeo, wheelMat);
      w.rotation.z = Math.PI/2; w.castShadow = true;
      const h = new THREE.Mesh(hubGeo, hubMat);
      h.rotation.z = Math.PI/2;
      g.add(w); g.add(h);
      return { pivot:g, spin:w };
    }
    const wheelPositions = [
      [-1.08, 0.44, 1.55], [1.08, 0.44, 1.55],   // front L, R
      [-1.08, 0.44, -1.55], [1.08, 0.44, -1.55]  // rear L, R
    ];
    const wheels = wheelPositions.map((p,idx)=>{
      const steerPivot = new THREE.Group();
      steerPivot.position.set(p[0],p[1],p[2]);
      const w = makeWheel();
      steerPivot.add(w.pivot);
      car.add(steerPivot);
      return { steerPivot, spin:w.spin, isFront: idx < 2 };
    });


    // angular diamond headlights
    [[-0.72,0.4,3.0],[0.72,0.4,3.0]].forEach(p=>{
      const l = new THREE.Mesh(new THREE.BoxGeometry(0.26,0.07,0.14), lightMat);
      l.position.set(p[0],p[1],p[2]);
      l.rotation.z = p[0] < 0 ? 0.5 : -0.5;
      car.add(l);
    });


    // Y-shaped taillights, each built from three thin bars
    const tailMatOwn = tailMat.clone(); // each car gets its own material instance so brake glow doesn't cross-affect other cars
    const tailLights = [];
    [-0.85, 0.85].forEach(x=>{
      const stem = new THREE.Mesh(new THREE.BoxGeometry(0.06,0.32,0.06), tailMatOwn);
      stem.position.set(x, 0.55, -2.28);
      const armA = new THREE.Mesh(new THREE.BoxGeometry(0.06,0.22,0.06), tailMatOwn);
      armA.position.set(x - (x<0?-0.12:0.12), 0.72, -2.26); armA.rotation.z = x<0 ? 0.6 : -0.6;
      const armB = new THREE.Mesh(new THREE.BoxGeometry(0.06,0.22,0.06), tailMatOwn);
      armB.position.set(x - (x<0?-0.12:0.12), 0.4, -2.26); armB.rotation.z = x<0 ? -0.6 : 0.6;
      car.add(stem, armA, armB);
      tailLights.push(stem, armA, armB);
    });


    scene.add(car);
    return {
      group: car, wheels, tailMat: tailMatOwn, bodyMat,
      bodyMeshes: [tub, nose, cabin, windshield, roofTrim, rearDeck, engineLouvers, wing, strutL, strutR, diffuser, ...tailLights]
    };
  }


  // =================================================================
  // Cars: player + 4 NPCs, staggered grid start
  // =================================================================
  // Lamborghini-inspired palette: Arancio orange, Verde lime, Giallo yellow, Blu, Grigio.
  const CAR_COLORS = [0xff5a12, 0x8ee81a, 0xf7d117, 0x1c6fe0, 0x6a7078];
  const MAX_OPPONENTS = 15;
  const startIdx = ((0 - 8) % SEGMENTS + SEGMENTS) % SEGMENTS;


  function spawnFor(slot){
    // slot 0 = player, 1..15 = opponents — staggered two-by-two racing grid behind the line
    const row = Math.floor(slot/2);
    const col = slot%2 === 0 ? -1 : 1;
    const idx = ((startIdx - row*4) % SEGMENTS + SEGMENTS) % SEGMENTS;
    const n = trackNormal(idx);
    const p = trackPts[idx].clone().addScaledVector(n, col*3.4);

  // every countdown, so restarting always starts clean rather than carrying over old speed/position.
  function resetRace(opponentCount){
    const pSpawn = spawnFor(0);
    forwardVel = 0; lateralVel = 0; prevForwardVel = 0;
    gearIndex = 1; heading = pSpawn.heading;
    bodyRoll = 0; bodyPitch = 0; slipAngle = 0; isDrifting = false;
    driftSessionPts = 0; totalDriftScore = 0;
    crashCooldown = 0; camShakeTime = 0; nitroWasActive = false;
    playerCar.group.position.copy(pSpawn.pos);
    playerCar.group.rotation.set(0, pSpawn.heading, 0);
    playerTrackIdx = pSpawn.idx;
    playerPrevIdx = pSpawn.idx;
    playerLapCount = 0;
    playerDist = cumLen[pSpawn.idx];
    playerLap = 0;
    raceFinished = false;
    raceFinishedHandled = false;
    finishTime = 0;
    finalPlace = null;


    activeNpcs = npcPool.slice(0, opponentCount);
    npcPool.forEach((npc, i)=>{
      const active = i < opponentCount;
      npc.obj.group.visible = active;
      if(!active) return;
      const sp = spawnFor(i+1);
      npc.pos.copy(sp.pos);
      npc.heading = sp.heading;
      npc.forwardVel = 0; npc.lateralVel = 0; npc.prevForwardVel = 0;
      npc.gearIndex = 1; npc.steerSmooth = 0; npc.roll = 0;
      npc.trackIdx = sp.idx;
      npc.prevIdx = sp.idx;
      npc.lapCount = 0;
      npc.distTraveled = cumLen[sp.idx];
      npc.obj.group.position.copy(sp.pos);
      npc.obj.group.rotation.set(0, sp.heading, 0);
    });
  }

  function applyAiDifficulty(){
    const targetMph = Number(aiSpeedSlider.value);
    const pct = (targetMph / 2.237) / TOP_SPEED_MS; // convert the chosen mph cap into a fraction of the player's top speed
    activeNpcs.forEach(npc=>{
      const scale = pct * (0.94 + Math.random()*0.1); // tight variance — these opponents are consistently fast
      npc.gears = {};
      Object.keys(GEARS).forEach(k=>{
        npc.gears[k] = { accel: GEARS[k].accel * scale, max: GEARS[k].max * scale };
      });
    });
  }

      el.textContent = seq[i];
      el.classList.remove('pop');
      void el.offsetWidth; // restart the pop animation
      el.classList.add('pop');
      playCountdownBeep(i === seq.length-1);

      }
    }
    step();
  }

  function findNearestIdx(pos, prevIdx){
    let best = prevIdx, bestDist = Infinity;
    for(let d=-20; d<=32; d++){
      const idx = ((prevIdx + d) % SEGMENTS + SEGMENTS) % SEGMENTS;
      const dx = trackPts[idx].x - pos.x, dz = trackPts[idx].z - pos.z;
      const dist = dx*dx + dz*dz;
      if(dist < bestDist){ bestDist = dist; best = idx; }
    }
    return best;
  }
  function normalizeAngle(a){
    while(a > Math.PI) a -= Math.PI*2;
    while(a < -Math.PI) a += Math.PI*2;
    return a;
  }


  // =================================================================
  // Sound engine — fully procedural via Web Audio API (engine hum, tire
  // screech on drift, brake screech, gear-shift click, background music).
  // No audio files.
  // =================================================================
  let audioCtx=null, engineOsc1,engineOsc2,engineFilter,engineGain;
  let noiseBuffer, driftSource, driftFilter, driftGain, brakeSource, brakeFilter, brakeGain;
  let musicGain=null, musicInterval=null, musicStep=0;


  function initAudio(){
    if(audioCtx) return;
    audioCtx = new (window.AudioContext||window.webkitAudioContext)();
    if(audioCtx.state === 'suspended') audioCtx.resume();


    engineOsc1 = audioCtx.createOscillator(); engineOsc1.type = 'sawtooth';
    engineOsc2 = audioCtx.createOscillator(); engineOsc2.type = 'square';
    engineFilter = audioCtx.createBiquadFilter(); engineFilter.type = 'lowpass'; engineFilter.frequency.value = 600;
    engineGain = audioCtx.createGain(); engineGain.gain.value = 0.001;
    engineOsc1.connect(engineFilter); engineOsc2.connect(engineFilter);
    engineFilter.connect(engineGain); engineGain.connect(audioCtx.destination);
    engineOsc1.frequency.value = 55; engineOsc2.frequency.value = 82;
    engineOsc1.start(); engineOsc2.start();


    const bufSize = audioCtx.sampleRate * 2;
    noiseBuffer = audioCtx.createBuffer(1, bufSize, audioCtx.sampleRate);
    const data = noiseBuffer.getChannelData(0);
    for(let i=0;i<bufSize;i++) data[i] = Math.random()*2-1;


    driftSource = audioCtx.createBufferSource(); driftSource.buffer = noiseBuffer; driftSource.loop = true;
    driftFilter = audioCtx.createBiquadFilter(); driftFilter.type = 'bandpass'; driftFilter.frequency.value = 2000; driftFilter.Q.value = 0.9;
    driftGain = audioCtx.createGain(); driftGain.gain.value = 0;
    driftSource.connect(driftFilter); driftFilter.connect(driftGain); driftGain.connect(audioCtx.destination);
    driftSource.start();


    brakeSource = audioCtx.createBufferSource(); brakeSource.buffer = noiseBuffer; brakeSource.loop = true;
    brakeFilter = audioCtx.createBiquadFilter(); brakeFilter.type = 'bandpass'; brakeFilter.frequency.value = 3400; brakeFilter.Q.value = 1.4;
    brakeGain = audioCtx.createGain(); brakeGain.gain.value = 0;
    brakeSource.connect(brakeFilter); brakeFilter.connect(brakeGain); brakeGain.connect(audioCtx.destination);
    brakeSource.start();


    // ignition rev blip
    const t0 = audioCtx.currentTime;
    engineGain.gain.setValueAtTime(0.001, t0);
    engineGain.gain.linearRampToValueAtTime(0.3, t0+0.15);
    engineGain.gain.linearRampToValueAtTime(0.09, t0+0.6);
    engineOsc1.frequency.setValueAtTime(55, t0);
    engineOsc1.frequency.linearRampToValueAtTime(230, t0+0.15);
    engineOsc1.frequency.linearRampToValueAtTime(80, t0+0.6);


    setMusicTheme(selectedMapIndex); // start with whichever circuit is currently selected
    startMusic();
  }


  // Background music: a simple procedural bassline + arpeggio loop per map, driven by
  // a plain JS interval rather than a precision Web Audio scheduler — good enough for
  // ambient menu/race music without the complexity of a full lookahead sequencer.
  // (MUSIC_THEMES itself is declared near the top of the script — buildWorld() needs
  // it during the very first, synchronous buildWorld(0) call before this section runs.)


  function playBassNote(freq, wave){
    const now = audioCtx.currentTime;
    const o = audioCtx.createOscillator(); o.type = wave || 'sawtooth'; o.frequency.value = freq;
    const f = audioCtx.createBiquadFilter(); f.type = 'lowpass'; f.frequency.value = 480;
    const g = audioCtx.createGain();
    g.gain.setValueAtTime(0.001, now);
    g.gain.linearRampToValueAtTime(0.5, now+0.02);
    g.gain.exponentialRampToValueAtTime(0.01, now+0.22);
    o.connect(f); f.connect(g); g.connect(musicGain);
    o.start(now); o.stop(now+0.24);
  }
  function playArpNote(freq, wave){
    const now = audioCtx.currentTime;
    const o = audioCtx.createOscillator(); o.type = wave || 'triangle'; o.frequency.value = freq;
    const g = audioCtx.createGain();
    g.gain.setValueAtTime(0.001, now);
    g.gain.linearRampToValueAtTime(0.22, now+0.015);
    g.gain.exponentialRampToValueAtTime(0.01, now+0.32);
    o.connect(g); g.connect(musicGain);
    o.start(now); o.stop(now+0.34);
  }
  function playHandDrum(){
    const now = audioCtx.currentTime;
    const src = audioCtx.createBufferSource(); src.buffer = noiseBuffer;
    const bp = audioCtx.createBiquadFilter(); bp.type = 'bandpass'; bp.frequency.value = 1800; bp.Q.value = 2.2;
    const g = audioCtx.createGain();
    g.gain.setValueAtTime(0.0001, now);
    g.gain.linearRampToValueAtTime(0.32, now+0.005);
    g.gain.exponentialRampToValueAtTime(0.001, now+0.09);
    src.connect(bp); bp.connect(g); g.connect(musicGain);
    src.start(now); src.stop(now+0.1);
  }
  function musicTick(){
    const t = currentMusicTheme;
    playBassNote(t.bass[musicStep % t.bass.length], t.waveBass);
    if(musicStep % 2 === 0) playArpNote(t.arp[(musicStep/2) % t.arp.length], t.waveArp);
    if(t.percussive && musicStep % 2 === 1) playHandDrum();
    musicStep++;
  }
  function startMusic(){
    if(musicInterval) return;
    musicGain = audioCtx.createGain();
    musicGain.gain.value = 0.8; // 5x louder per feedback — it needs to actually be heard
    musicGain.connect(audioCtx.destination);
    musicInterval = setInterval(musicTick, currentMusicTheme.tempoMs);
  }
  // Swaps the playing theme — restarts the interval at the new theme's tempo. Safe to
  // call before audio is unlocked (just updates which theme is queued up for later).
  function setMusicTheme(mapIndex){
    const theme = MUSIC_THEMES[mapIndex] || MUSIC_THEMES[0];
    if(theme === currentMusicTheme) return;
    currentMusicTheme = theme;
    musicStep = 0;
    if(musicInterval){
      clearInterval(musicInterval);
      musicInterval = setInterval(musicTick, theme.tempoMs);
    }
  }


  // Crowd cheering — a noise-based "roar" swell (bandpass-filtered noise with a
  // wobbling filter frequency for texture) since we have no vocal samples to draw on.
  // Used for the race start, every completed lap, and the finish, scaled by placement.
  function playCrowdCheer(intensity, duration){
    if(!audioCtx || !noiseBuffer) return;
    const now = audioCtx.currentTime;
    const src = audioCtx.createBufferSource(); src.buffer = noiseBuffer; src.loop = true;
    const bp = audioCtx.createBiquadFilter(); bp.type = 'bandpass'; bp.frequency.value = 900; bp.Q.value = 0.6;
    const lp = audioCtx.createBiquadFilter(); lp.type = 'lowpass'; lp.frequency.value = 2400;
    const g = audioCtx.createGain();
    g.gain.setValueAtTime(0.0001, now);
    g.gain.linearRampToValueAtTime(0.55*intensity, now + duration*0.22);
    g.gain.linearRampToValueAtTime(0.4*intensity, now + duration*0.6);
    g.gain.exponentialRampToValueAtTime(0.001, now + duration);
    src.connect(bp); bp.connect(lp); lp.connect(g); g.connect(audioCtx.destination);
    src.start(now); src.stop(now + duration + 0.1);
    // wobble the bandpass center frequency so the roar has some organic movement
    const lfo = audioCtx.createOscillator(); lfo.frequency.value = 4 + Math.random()*3;
    const lfoGain = audioCtx.createGain(); lfoGain.gain.value = 260;
    lfo.connect(lfoGain); lfoGain.connect(bp.frequency);
    lfo.start(now); lfo.stop(now + duration + 0.1);
  }

  function playCrashSound(intensity){
    if(!audioCtx) return;
    const now = audioCtx.currentTime;
    const amt = THREE.MathUtils.clamp(intensity, 0.3, 1);
    // noise burst (crunch)
    const src = audioCtx.createBufferSource(); src.buffer = noiseBuffer;
    const filter = audioCtx.createBiquadFilter(); filter.type = 'lowpass'; filter.frequency.value = 1400;
    const g = audioCtx.createGain();
    g.gain.setValueAtTime(0.55*amt, now);
    g.gain.exponentialRampToValueAtTime(0.001, now+0.35);
    src.connect(filter); filter.connect(g); g.connect(audioCtx.destination);
    src.start(now); src.stop(now+0.36);
    // low thud (impact)
    const o = audioCtx.createOscillator(); o.type = 'sine';
    o.frequency.setValueAtTime(95, now);
    o.frequency.exponentialRampToValueAtTime(38, now+0.22);
    const og = audioCtx.createGain();
    og.gain.setValueAtTime(0.55*amt, now);
    og.gain.exponentialRampToValueAtTime(0.001, now+0.25);
    o.connect(og); og.connect(audioCtx.destination);
    o.start(now); o.stop(now+0.26);
  }


  function updateAudio(){
    if(!audioCtx) return;
    const g = gearIndex !== 0 ? GEARS[String(gearIndex)] : null;
    const rpmFrac = g ? THREE.MathUtils.clamp(Math.abs(forwardVel)/g.max, 0.06, 1.15) : 0.12;
    const now = audioCtx.currentTime;
    const baseFreq = 45 + rpmFrac*260;
    engineOsc1.frequency.setTargetAtTime(baseFreq, now, 0.05);
    engineOsc2.frequency.setTargetAtTime(baseFreq*1.5, now, 0.05);
    engineFilter.frequency.setTargetAtTime(350 + rpmFrac*2600, now, 0.05);
    const targetGain = 0.05 + (keys.up ? 0.14 : 0.02) + rpmFrac*0.07;
    engineGain.gain.setTargetAtTime(targetGain, now, 0.09);


    const driftIntensity = isDrifting ? THREE.MathUtils.clamp(Math.abs(slipAngle)*1.8, 0, 1) : 0;
    driftGain.gain.setTargetAtTime(driftIntensity*0.32, now, 0.06);
    driftFilter.frequency.setTargetAtTime(1700 + driftIntensity*1400, now, 0.06);


    const brakeIntensity = (keys.down && Math.abs(forwardVel) > 7) ? THREE.MathUtils.clamp(Math.abs(forwardVel)/30, 0, 1) : 0;
    brakeGain.gain.setTargetAtTime(brakeIntensity*0.22, now, 0.06);
  }


  // Web Audio's setTargetAtTime schedules an automation that keeps running on the audio
  // clock even after we stop calling updateAudio() — so once the race freezes (started
  // = false), whatever engine/drift/brake volume was last set would otherwise sustain
  // indefinitely. Explicitly cut everything back to near-silent instead.
  function silenceAudio(){
    if(!audioCtx) return;
    const now = audioCtx.currentTime;
    // Linear-ramp every gain to true zero on a fixed, short timeline — an exponential
    // "setTargetAtTime" approach never actually reaches zero and can leave an audible
    // whine (especially right after a nitro boost, when pitch is highest) hanging
    // around for way longer than it should.
    [engineGain, driftGain, brakeGain].forEach(g=>{
      const current = g.gain.value;
      g.gain.cancelScheduledValues(now);
      g.gain.setValueAtTime(current, now);
      g.gain.linearRampToValueAtTime(0.0001, now + 0.35);
    });
  }


  // =================================================================
  // Physics update
  // =================================================================
  function updatePlayerPhysics(dt){
    const throttle = keys.up;
    const brake = keys.down;
    prevForwardVel = forwardVel;


    // Mobile players wanted a much faster car — scale accel/top-speed by 4x, and ease
    // off aerodynamic drag to match so the car can actually reach that speed instead
    // of just hitting a wall of drag partway there. AI and PC mode are unaffected.
    const mult = (controlMode === 'mobile') ? 4 : 1;


    const nitroCombo = throttle && brake; // press throttle + brake together: 2x acceleration boost
    if(nitroCombo){
      if(gearIndex <= 0){ gearIndex = 5; updateGearUI(); } // combo needs a forward gear engaged
      if(forwardVel < TOP_SPEED_MS*mult){
        const g = GEARS[String(Math.max(1, gearIndex))];
        forwardVel = Math.min(TOP_SPEED_MS*mult, forwardVel + g.accel * 2 * mult * dt);
      }
      if(!nitroWasActive) showToast('NITRO!');
      nitroWasActive = true;
      lateralVel *= 0.85;
    } else {
      nitroWasActive = false;
      if(gearIndex !== 0){
        const g = GEARS[String(gearIndex)];
        const forwardGear = gearIndex > 0;
        if(throttle && !brake){
          if(forwardGear && forwardVel < g.max*mult) forwardVel += g.accel * mult * dt;
          if(!forwardGear && forwardVel > -g.max*mult) forwardVel -= g.accel * mult * dt;
        } else if(!brake){
          // engine braking: coasting off-throttle slows you faster in a low gear, like real compression braking
          const engineBrake = forwardGear ? (6 - gearIndex) * 0.35 : 1.2;
          forwardVel -= Math.sign(forwardVel) * engineBrake * mult * dt;
          if(Math.sign(prevForwardVel) !== Math.sign(forwardVel)) forwardVel = 0;
        }
      }
      if(brake){
        if(forwardVel > 0) forwardVel = Math.max(0, forwardVel - BRAKE_DECEL*mult*dt);
        else if(forwardVel < 0) forwardVel = Math.min(0, forwardVel + BRAKE_DECEL*mult*dt);
      }
      // rolling resistance (always on, gentle)
      forwardVel -= forwardVel * Math.min(1, ROLL_DRAG*dt);
      // aerodynamic drag — grows with the square of speed, so it bites hardest near top speed
      forwardVel -= Math.sign(forwardVel) * (AERO_DRAG/mult) * forwardVel * forwardVel * dt;
      if(Math.abs(forwardVel) < 0.02) forwardVel = 0;
    }


    // steering: full authority at moderate speed, tapers off at high speed like real power steering
    const lowSpeedFactor = THREE.MathUtils.clamp(Math.abs(forwardVel)/8, 0.35, 1);
    const highSpeedTaper = THREE.MathUtils.clamp(1 - Math.max(0, Math.abs(forwardVel)-22)/70, 0.4, 1);
    const speedFactor = lowSpeedFactor * highSpeedTaper;
    const dir = forwardVel >= 0 ? 1 : -1;
    let steerInput = 0;
    if(keys.left) steerInput -= 1;
    if(keys.right) steerInput += 1;
    const turnRate = STEER_RATE * steerInput * speedFactor * dir;
    heading -= turnRate * dt;


    // lateral slip / tire grip model — this is what produces real drifting
    lateralVel += -turnRate * forwardVel * dt * 0.3; // scaled down since STEER_RATE doubled — keeps drift feel similar
    const gripRecovery = brake ? 3.2 : 6.5; // tires grip back slower mid-brake, easier to hold a drift
    lateralVel -= lateralVel * Math.min(1, gripRecovery*dt);


    // sliding scrubs off forward speed, same as a real car losing momentum to a slide
    forwardVel -= Math.sign(forwardVel) * Math.abs(lateralVel) * 0.08 * dt;


    const forward = new THREE.Vector3(Math.sin(heading), 0, Math.cos(heading));
    const right = new THREE.Vector3(Math.cos(heading), 0, -Math.sin(heading));
    const vel = forward.clone().multiplyScalar(forwardVel).addScaledVector(right, lateralVel);


    playerCar.group.position.addScaledVector(vel, dt);
    playerCar.group.rotation.y = heading;


    // weight transfer: body rolls into slides/turns and pitches under accel & braking
    const targetRoll = THREE.MathUtils.clamp(turnRate * forwardVel * 0.012, -0.12, 0.12);
    const accelThisFrame = (forwardVel - prevForwardVel) / Math.max(dt, 0.001);
    const targetPitch = THREE.MathUtils.clamp(-accelThisFrame * 0.006, -0.07, 0.09);
    bodyRoll += (targetRoll - bodyRoll) * Math.min(1, 9*dt);
    bodyPitch += (targetPitch - bodyPitch) * Math.min(1, 7*dt);
    playerCar.group.rotation.z = bodyRoll;
    playerCar.group.rotation.x = bodyPitch;


    // front wheel steering visual + wheel spin
    playerCar.wheels.forEach(w=>{
      if(w.isFront) w.steerPivot.rotation.y = steerInput * 0.5;
      w.spin.rotation.x -= forwardVel*dt*2.4;
    });
    playerCar.tailMat.emissiveIntensity = brake ? 1.8 : 0.6;


    // drift detection + scoring
    slipAngle = Math.abs(forwardVel) > 3 ? Math.atan2(lateralVel, Math.abs(forwardVel)) : 0;
    const wasDrifting = isDrifting;
    isDrifting = Math.abs(slipAngle) > 0.22 && Math.abs(forwardVel) > 5;
    if(isDrifting){
      driftSessionPts += Math.abs(slipAngle) * Math.abs(forwardVel) * dt * 14;
    } else if(wasDrifting){
      if(driftSessionPts > 15){
        totalDriftScore += Math.round(driftSessionPts);
        showToast('+' + Math.round(driftSessionPts) + ' DRIFT!');
      }
      driftSessionPts = 0;
    }


    // track progress — recomputed fresh from actual position every frame (not
    // integrated from velocity), so a collision or wall bounce can never desync
    // this from where the car really is on track.
    playerTrackIdx = findNearestIdx(playerCar.group.position, playerTrackIdx);
    if(playerPrevIdx - playerTrackIdx > SEGMENTS*0.5){
      playerLapCount++;   // wrapped forward across the line: idx dropped from near-end to near-start
      if(playerLapCount > 0 && playerLapCount <= TOTAL_LAPS) playCrowdCheer(0.55, 1.2); // crowd cheers each completed lap
    }
    else if(playerTrackIdx - playerPrevIdx > SEGMENTS*0.5) playerLapCount--; // drove backward across the line
    playerPrevIdx = playerTrackIdx;
    playerDist = playerLapCount*TOTAL_LEN + cumLen[playerTrackIdx];
    playerLap = playerLapCount;


    // crash mechanics: hitting the guardrail kills speed, bounces you back, and kicks the car sideways
    crashCooldown = Math.max(0, crashCooldown - dt);
    const n = trackNormal(playerTrackIdx);
    const centerToCar = playerCar.group.position.clone().sub(trackPts[playerTrackIdx]);
    const lateralOffset = centerToCar.dot(n);
    const wallLimit = TRACK_WIDTH/2 + 0.5;
    if(Math.abs(lateralOffset) > wallLimit){
      const impactSpeed = Math.abs(forwardVel) + Math.abs(lateralVel);
      const over = Math.abs(lateralOffset) - wallLimit;
      const sign = Math.sign(lateralOffset);
      // clamp back to the wall
      playerCar.group.position.addScaledVector(n, -sign*over);
      if(crashCooldown <= 0 && impactSpeed > 3){
        const intensity = THREE.MathUtils.clamp(impactSpeed/40, 0.3, 1);
        forwardVel *= 0.4;
        lateralVel = -sign * impactSpeed * 0.18;
        heading += (Math.random()-0.5) * 0.25 * intensity;
        crashCooldown = 0.6;
        camShakeTime = 0.35 * intensity;
        playCrashSound(intensity);
        showToast('CRASH!', true);
      } else {
        lateralVel *= 0.2; // still pinned against the wall, just bleed the sideways motion
      }
    }


    if(!raceFinished && playerLap >= TOTAL_LAPS){
      raceFinished = true;
      finishTime = (performance.now() - raceStartTime)/1000;
    }


    return { pos: playerCar.group.position, distTraveled: playerDist };
  }


  // car-vs-car collisions: simple circle-based impulse response between the player and each NPC
  const CAR_COLLIDE_RADIUS = 2.9;
  function resolveCarCollisions(dt){
    activeNpcs.forEach(npc=>{
      const dx = playerCar.group.position.x - npc.pos.x;
      const dz = playerCar.group.position.z - npc.pos.z;
      const dist = Math.hypot(dx, dz);
      if(dist < CAR_COLLIDE_RADIUS && dist > 0.001){
        const nx = dx/dist, nz = dz/dist;
        const overlap = CAR_COLLIDE_RADIUS - dist;
        playerCar.group.position.x += nx*overlap*0.6;
        playerCar.group.position.z += nz*overlap*0.6;
        npc.pos.x -= nx*overlap*0.4;
        npc.pos.z -= nz*overlap*0.4;


        const rel = Math.abs(forwardVel) + npc.forwardVel;
        if(crashCooldown <= 0 && rel > 4){
          const intensity = THREE.MathUtils.clamp(rel/40, 0.3, 1);
          const right = new THREE.Vector3(Math.cos(heading), 0, -Math.sin(heading));
          const lateralPush = (nx*right.x + nz*right.z) * rel * 0.22;
          lateralVel += lateralPush;
          forwardVel *= 0.65;
          npc.forwardVel *= 0.55;
          npc.heading += (Math.random()-0.5) * 0.3;
          crashCooldown = 0.6;
          camShakeTime = 0.3 * intensity;
          playCrashSound(intensity);
          showToast('CRASH!', true);
        }
      }
    });
  }


  function updateNpc(npc, dt){
    // --- decide inputs the way a human racer would: look ahead, steer toward the
    // apex, and use the brake pedal (not just less throttle) before sharp corners ---
    const LOOKAHEAD = 16;
    const targetIdx = (npc.trackIdx + LOOKAHEAD) % SEGMENTS;
    const target = trackPts[targetIdx];
    const dx = target.x - npc.pos.x, dz = target.z - npc.pos.z;
    const desiredHeading = Math.atan2(dx, dz);
    const diff = normalizeAngle(desiredHeading - npc.heading);
    // Note the negative sign: with this car's forward = (sin(heading), cos(heading))
    // convention, a positive diff means heading needs to INCREASE to close the gap,
    // but the shared "heading -= turnRate*dt" integration (same as the player's) expects
    // a NEGATIVE steer value to increase heading — so we flip it here.
    const steerTarget = THREE.MathUtils.clamp(-diff * npc.turnGain, -1, 1);
    // human-style reaction smoothing instead of snapping the wheel instantly — quick reflexes
    npc.steerSmooth += (steerTarget - npc.steerSmooth) * Math.min(1, 11*dt);
    const steerInput = npc.steerSmooth;


    const severity = Math.abs(diff);
    const g = npc.gears;
    const cornerSpeedCap = g['5'].max * THREE.MathUtils.clamp(1 - severity*0.95, 0.42, 1);
    let throttle = true, brake = false;
    if(npc.forwardVel > cornerSpeedCap + 6){ throttle = false; brake = true; }
    else if(npc.forwardVel > cornerSpeedCap){ throttle = false; brake = false; }


    // --- same physics model the player uses: gears, drag, tire slip, weight transfer ---
    npc.gearIndex = autoGearFor(npc.forwardVel, g);
    npc.prevForwardVel = npc.forwardVel;


    if(npc.gearIndex !== 0){
      const gr = g[String(npc.gearIndex)];
      if(throttle && !brake){
        if(npc.forwardVel < gr.max) npc.forwardVel += gr.accel*dt;
      } else if(!brake){
        const engineBrake = (6 - npc.gearIndex) * 0.35;
        npc.forwardVel = Math.max(0, npc.forwardVel - engineBrake*dt);
      }
    }
    if(brake) npc.forwardVel = Math.max(0, npc.forwardVel - BRAKE_DECEL*0.85*dt);


    npc.forwardVel -= npc.forwardVel * Math.min(1, ROLL_DRAG*dt);
    npc.forwardVel -= AERO_DRAG * npc.forwardVel * npc.forwardVel * dt;
    if(npc.forwardVel < 0.02) npc.forwardVel = 0;


    const lowSpeedFactor = THREE.MathUtils.clamp(npc.forwardVel/8, 0.35, 1);
    const highSpeedTaper = THREE.MathUtils.clamp(1 - Math.max(0, npc.forwardVel-22)/70, 0.4, 1);
    const speedFactor = lowSpeedFactor * highSpeedTaper;
    const turnRate = STEER_RATE * steerInput * speedFactor;
    npc.heading -= turnRate * dt;


    npc.lateralVel += -turnRate * npc.forwardVel * dt * 0.3;
    const gripRecovery = brake ? 3.2 : 6.5;
    npc.lateralVel -= npc.lateralVel * Math.min(1, gripRecovery*dt);
    npc.forwardVel = Math.max(0, npc.forwardVel - Math.abs(npc.lateralVel)*0.08*dt);


    const forward = new THREE.Vector3(Math.sin(npc.heading), 0, Math.cos(npc.heading));
    const right = new THREE.Vector3(Math.cos(npc.heading), 0, -Math.sin(npc.heading));
    const vel = forward.clone().multiplyScalar(npc.forwardVel).addScaledVector(right, npc.lateralVel);
    npc.pos.addScaledVector(vel, dt);


    // weight transfer, same as the player's car
    const targetRoll = THREE.MathUtils.clamp(turnRate * npc.forwardVel * 0.012, -0.12, 0.12);
    npc.roll += (targetRoll - npc.roll) * Math.min(1, 9*dt);


    npc.obj.group.position.copy(npc.pos);
    npc.obj.group.rotation.y = npc.heading;
    npc.obj.group.rotation.z = npc.roll;
    npc.obj.wheels.forEach(w=>{
      if(w.isFront) w.steerPivot.rotation.y = steerInput*0.5;
      w.spin.rotation.x -= npc.forwardVel*dt*2.4;
    });
    npc.obj.tailMat.emissiveIntensity = brake ? 1.8 : 0.6;


    // keep NPCs off the guardrails too, same boundary the player has
    npc.trackIdx = findNearestIdx(npc.pos, npc.trackIdx);
    const n = trackNormal(npc.trackIdx);
    const centerToCar = npc.pos.clone().sub(trackPts[npc.trackIdx]);
    const lateralOffset = centerToCar.dot(n);
    const wallLimit = TRACK_WIDTH/2 - 0.4;
    if(Math.abs(lateralOffset) > wallLimit){
      const over = Math.abs(lateralOffset) - wallLimit;
      const sign = Math.sign(lateralOffset);
      npc.pos.addScaledVector(n, -sign*over);
      npc.lateralVel *= 0.25;
      npc.forwardVel *= 0.85;
    }


    // progress recomputed fresh from actual position every frame — see the matching
    // comment in updatePlayerPhysics for why this replaced velocity integration.
    if(npc.prevIdx - npc.trackIdx > SEGMENTS*0.5) npc.lapCount++;   // wrapped forward across the line
    else if(npc.trackIdx - npc.prevIdx > SEGMENTS*0.5) npc.lapCount--; // drove backward across the line
    npc.prevIdx = npc.trackIdx;
    npc.distTraveled = npc.lapCount*TOTAL_LEN + cumLen[npc.trackIdx];
  }

  // =================================================================
  function updateHUD(){
    const mph = Math.round(Math.abs(forwardVel) * 2.237);
    document.getElementById('speed-val').textContent = String(mph).padStart(3,'0');


    const g = gearIndex !== 0 ? GEARS[String(gearIndex)] : null;
    const rpmFrac = g ? THREE.MathUtils.clamp(Math.abs(forwardVel)/g.max, 0.08, 1) : 0.1;
    document.getElementById('rpm-bar').style.width = (rpmFrac*100).toFixed(0)+'%';


    // cockpit dash
    document.getElementById('cockpit-speed-val').innerHTML = mph + '<small>MPH</small>';
    document.querySelector('#gauge-speed .ring').style.setProperty('--val', Math.min(1, mph/350));
    document.querySelector('#gauge-rpm .ring').style.setProperty('--val', rpmFrac);
    document.getElementById('gear-readout').textContent = gearLabel(gearIndex);


    let steerInput = 0;
    if(keys.left) steerInput -= 1;
    if(keys.right) steerInput += 1;
    document.getElementById('wheel').style.transform = 'rotate(' + (steerInput*110) + 'deg)';


    // race position — frozen at the moment you finish, so opponents completing their
    // remaining laps afterward can no longer change your final result
    const totalRacers = activeNpcs.length + 1;
    let place;
    if(raceFinished && finalPlace !== null){
      place = finalPlace;
    } else {
      const all = [{ dist: playerDist, isPlayer:true }];
      activeNpcs.forEach(n=> all.push({ dist:n.distTraveled, isPlayer:false }));
      all.sort((a,b)=> b.dist - a.dist);
      place = all.findIndex(a=>a.isPlayer) + 1;
    }
    const ord = ordinal(place);
    document.getElementById('race-pos').textContent = ord + ' / ' + totalRacers;
    document.getElementById('race-lap').textContent = 'LAP ' + Math.min(TOTAL_LAPS, Math.max(1,playerLap+1)) + ' / ' + TOTAL_LAPS;


    if(started){
      const elapsed = raceFinished ? finishTime : (performance.now()-raceStartTime)/1000;
      const mm = Math.floor(elapsed/60).toString().padStart(2,'0');
      const ss = (elapsed%60).toFixed(1).padStart(4,'0');
      document.getElementById('race-time').textContent = mm+':'+ss;
    }


    if(raceFinished){
      if(finalPlace === null){
        // freeze the placement right now, at the instant you crossed the line
        const all = [{ dist: playerDist, isPlayer:true }];
        activeNpcs.forEach(n=> all.push({ dist:n.distTraveled, isPlayer:false }));
        all.sort((a,b)=> b.dist - a.dist);
        finalPlace = all.findIndex(a=>a.isPlayer) + 1;
      }
      const banner = document.getElementById('finish-banner');
      banner.classList.add('show');
      const mm = Math.floor(finishTime/60).toString().padStart(2,'0');
      const ss = (finishTime%60).toFixed(1).padStart(4,'0');
      document.getElementById('finish-detail').textContent = ordinal(finalPlace) + ' PLACE  \u00b7  ' + mm+':'+ss;


      if(!raceFinishedHandled){
        raceFinishedHandled = true;
        started = false; // freeze every car exactly where the race ended
        silenceAudio(); // otherwise the engine/drift/brake sound holds at its last volume forever
        const cheerIntensity = Math.max(0.35, 1 - (finalPlace-1)*0.09);
        playCrowdCheer(cheerIntensity, finalPlace <= 3 ? 2.4 : 1.6); // the crowd goes wildest for a podium finish
        recordRaceResult(finalPlace, finishTime, totalDriftScore);
        if(inPartyChallenge){ postPartyResult(finalPlace, finishTime); inPartyChallenge = false; }
        setTimeout(returnToHome, 4500);
      }
    }


    // drift meter
    const driftBarWrap = document.getElementById('drift-bar-wrap');
    driftBarWrap.classList.toggle('active', isDrifting);
    document.getElementById('drift-bar').style.width = Math.min(100, driftSessionPts/2.5) + '%';
    document.getElementById('drift-angle').textContent = Math.round(Math.abs(slipAngle)*180/Math.PI) + '\u00b0 SLIP';
    document.getElementById('drift-total').textContent = 'DRIFT ' + totalDriftScore;


    updateAudio();
  }


  // =================================================================
  // Camera
  // =================================================================
  const camPosTarget = new THREE.Vector3();
  const camLookTarget = new THREE.Vector3();
  const currentLook = new THREE.Vector3();
  let lookInit = false;


  function updateCamera(dt){
    const forward = new THREE.Vector3(Math.sin(heading), 0, Math.cos(heading));
    const right = new THREE.Vector3(Math.cos(heading), 0, -Math.sin(heading));
    const carPos = playerCar.group.position;


    if(camMode === 0){ // CHASE
      camPosTarget.copy(carPos).addScaledVector(forward, -9).add(new THREE.Vector3(0,4.2,0));
      camLookTarget.copy(carPos).add(new THREE.Vector3(0,1.2,0));
      camera.position.lerp(camPosTarget, 1 - Math.pow(0.001, dt));
    } else if(camMode === 1){ // COCKPIT - eye point inside the car, body hidden so the road is fully visible
      camPosTarget.copy(carPos).addScaledVector(forward, 0.15).add(new THREE.Vector3(0,1.18,0));
      camLookTarget.copy(carPos).addScaledVector(forward, 12).add(new THREE.Vector3(0,1.1,0));
      camera.position.copy(camPosTarget);
    } else { // SIDE
      camPosTarget.copy(carPos).addScaledVector(right, 9).add(new THREE.Vector3(0,3.2,0));
      camLookTarget.copy(carPos).add(new THREE.Vector3(0,1.0,0));
      camera.position.lerp(camPosTarget, 1 - Math.pow(0.001, dt));
    }


    if(!lookInit){ currentLook.copy(camLookTarget); lookInit = true; }
    const lookLerp = camMode === 1 ? 1 : (1 - Math.pow(0.0005, dt));
    currentLook.lerp(camLookTarget, lookLerp);


    // crash camera shake
    if(camShakeTime > 0){
      camShakeTime = Math.max(0, camShakeTime - dt);
      const mag = camShakeTime * 0.5;
      camera.position.x += (Math.random()-0.5) * mag;
      camera.position.y += (Math.random()-0.5) * mag;
      const flash = document.getElementById('crash-flash');
      flash.style.opacity = Math.min(0.5, camShakeTime*1.3);
    } else {
      document.getElementById('crash-flash').style.opacity = 0;
    }


    camera.lookAt(currentLook);


    // sense of speed: FOV widens slightly as you go faster, like a real dashcam/helmet-cam effect
    const targetFov = 70 + THREE.MathUtils.clamp(Math.abs(forwardVel)*0.35, 0, 16);
    camera.fov += (targetFov - camera.fov) * Math.min(1, 4*dt);
    camera.updateProjectionMatrix();
  }


  // =================================================================
  // Main loop
  // =================================================================
  const clock = new THREE.Clock();
  function animate(){
    requestAnimationFrame(animate);
    const dt = Math.min(clock.getDelta(), 0.05);
    if(started){
      updatePlayerPhysics(dt);
      activeNpcs.forEach(n => updateNpc(n, dt));
      resolveCarCollisions(dt);
      updateHUD();
    }
    updateCamera(dt);
    renderer.render(scene, camera);
  }
  updateGearUI();
  updateCamUI();
  animate();


  // Now that playerCar, the paint booth, and everything else exists, resolve the
  // initial login state (auto-continue a saved session, or show the sign-in screen).
  (async ()=>{
    const existingUser = await getCurrentUser();
    const accounts = await loadAccounts();
    if(existingUser && accounts[existingUser]){
      hideLoginScreen();
      await refreshAccountBadge();
    } else {
      showLoginScreen();
    }
  })();
})();
